        remaining_ms = max(1000.0 * self.max_wallclock_seconds - elapsed_ms, 0.0)
        return remaining_ms / max(warmdown_ms, 1e-9) if remaining_ms <= warmdown_ms else 1.0

def token_chunks(total_tokens: int, seq_len: int, max_chunk_tokens: int) -> list[int]:
    usable_total = (total_tokens // seq_len) * seq_len
    if usable_total <= 0:
        raise ValueError(f"token budget too small for seq_len={seq_len}")
    usable_chunk = max((max_chunk_tokens // seq_len) * seq_len, seq_len)
    chunks: list[int] = []
    remaining = usable_total
    while remaining > 0:
        chunk = min(remaining, usable_chunk)
        chunks.append(chunk)
        remaining -= chunk
    return chunks


def accumulate_flat_grads(
    accum: dict[str, mx.array] | None,
    grads_tree: dict,
    scale: float,
) -> dict[str, mx.array]:
    flat = dict(tree_flatten(grads_tree))
    if accum is None:
        return {k: g * scale for k, g in flat.items()}
    for k, g in flat.items():
        accum[k] = accum[k] + g * scale
    return accum


def peak_rss_bytes() -> int:
    rss = int(resource.getrusage(resource.RUSAGE_SELF).ru_maxrss)
    # macOS reports bytes; Linux typically reports KiB.
    return rss if sys.platform == "darwin" else rss * 1024


def format_bytes(num_bytes: int) -> str:
    value = float(num_bytes)
    for unit in ("B", "KiB", "MiB", "GiB", "TiB"):
        if value < 1024.0 or unit == "TiB":
            return f"{value:.2f}{unit}"
        value /= 1024.0
    return f"{num_bytes}B"

def rms_norm(x: mx.array, eps: float = 1e-6) -> mx.array:
    return (x * mx.rsqrt(mx.mean(x * x, axis=-1, keepdims=True) + eps)).astype(x.dtype)


def zeropower_newtonschulz5(g: mx.array, steps: int, eps: float = 1e-7) -> mx.array:
    # Orthogonalize a 2D update matrix with a fast Newton-Schulz iteration.
    # Muon uses this to normalize matrix-shaped gradients before applying them.
    # Background on Muon: https://kellerjordan.github.io/posts/muon/
    a, b, c = 3.4445, -4.7750, 2.0315
    x = g.astype(mx.float32)
    x = x / (mx.sqrt(mx.sum(x * x)) + eps)
    transposed = x.shape[0] > x.shape[1]
    if transposed:
        x = x.T
    for _ in range(steps):
        a_mat = x @ x.T
        b_mat = b * a_mat + c * (a_mat @ a_mat)
        x = a * x + b_mat @ x
    if transposed:
        x = x.T
    return x.astype(g.dtype)


def load_data_shard(path: Path) -> np.ndarray:
    header_bytes = 256 * np.dtype("<i4").itemsize
    token_bytes = np.dtype("<u2").itemsize
    header = np.fromfile(path, dtype="<i4", count=256)
    if header.size != 256 or int(header[0]) != 20240520 or int(header[1]) != 1:
        raise ValueError(f"Unexpected shard header for {path}")
    num_tokens = int(header[2])
    if path.stat().st_size != header_bytes + num_tokens * token_bytes:
        raise ValueError(f"Shard size mismatch for {path}")
    tokens = np.fromfile(path, dtype="<u2", count=num_tokens, offset=header_bytes)
    if tokens.size != num_tokens:
        raise ValueError(f"Short read for {path}")
    return tokens.astype(np.int32, copy=False)

class TokenStream:
    def __init__(
        self,
        pattern: str,
        log_fn: Callable[[str], None] | None = None,
        dataset_name: str = "",
    ):
        self.files = [Path(p) for p in sorted(glob.glob(pattern))]
        if not self.files:
            raise FileNotFoundError(f"No files found for pattern: {pattern}")
        self.epoch = 1
        self.file_idx = 0
        self.log_fn = log_fn
        self.dataset_name = dataset_name
        self.tokens = load_data_shard(self.files[0])
        self.pos = 0

    def next_file(self) -> None:
        self.file_idx = (self.file_idx + 1) % len(self.files)
        if self.file_idx == 0:
            self.epoch += 1
            if self.log_fn is not None:
                self.log_fn(
                    f"WARNING: starting epoch:{self.epoch} "
                    f"dataset:{self.dataset_name} train_shards:{len(self.files)}"
                )
        self.tokens = load_data_shard(self.files[self.file_idx])
        self.pos = 0

    def take(self, n: int) -> np.ndarray:
        chunks: list[np.ndarray] = []
        left = n
        while left > 0:
            if self.pos >= self.tokens.size:
                self.next_file()
            k = min(left, int(self.tokens.size - self.pos))
            chunks.append(self.tokens[self.pos : self.pos + k])
            self.pos += k
            left -= k
        return chunks[0] if len(chunks) == 1 else np.concatenate(chunks, axis=0)


class TokenLoader:
    def __init__(
        self,
        pattern: str,
        log_fn: Callable[[str], None] | None = None,
        dataset_name: str = "",
    ):
        self.stream = TokenStream(pattern, log_fn=log_fn, dataset_name=dataset_name)

    def next_batch(self, batch_tokens: int, seq_len: int) -> tuple[mx.array, mx.array]:
        usable = (batch_tokens // seq_len) * seq_len
        if usable <= 0:
            raise ValueError(f"token budget too small for seq_len={seq_len}")
        chunk = self.stream.take(usable + 1)
        x = chunk[:-1].reshape(-1, seq_len)
        y = chunk[1:].reshape(-1, seq_len)
        return mx.array(x, dtype=mx.int32), mx.array(y, dtype=mx.int32)

_ACTIVE_ACTIVATION_COLLECTOR = None


class ActivationAbsMeanCollector:
    def __init__(self):
        self.sum_abs: dict[str, np.ndarray] = {}
        self.counts: dict[str, int] = {}

    def observe(self, name: str, x: mx.array) -> None:
        x_np = np.array(x.astype(mx.float32), dtype=np.float32, copy=False)
        if x_np.ndim == 0:
            return
        flat = x_np.reshape(-1, x_np.shape[-1])
        if flat.size == 0:
            return
        sum_abs = np.sum(np.abs(flat), axis=0, dtype=np.float64)
        prev = self.sum_abs.get(name)
        self.sum_abs[name] = sum_abs if prev is None else prev + sum_abs
        self.counts[name] = self.counts.get(name, 0) + int(flat.shape[0])

    def finalize(self) -> dict[str, np.ndarray]:
        out: dict[str, np.ndarray] = {}
        for name, sum_abs in self.sum_abs.items():
            out[name] = np.ascontiguousarray(
                (sum_abs / max(self.counts.get(name, 0), 1)).astype(np.float32, copy=False)
            )
        return out


class ActivationSecondMomentCollector:
    def __init__(
        self,
        group_size: int,
        max_rows_per_observe: int,
        transform_fn: Callable[[str, np.ndarray], np.ndarray] | None = None,
    ):
        self.group_size = max(group_size, 1)
        self.max_rows_per_observe = max_rows_per_observe
        self.transform_fn = transform_fn
        self.sum_xxt: dict[str, list[np.ndarray]] = {}
        self.counts: dict[str, int] = {}

    def observe(self, name: str, x: mx.array) -> None:
        x_np = np.array(x.astype(mx.float32), dtype=np.float32, copy=False)
        if x_np.ndim == 0:
            return
        flat = x_np.reshape(-1, x_np.shape[-1])
        if flat.size == 0:
            return
        if self.max_rows_per_observe > 0 and flat.shape[0] > self.max_rows_per_observe:
            stride = max(flat.shape[0] // self.max_rows_per_observe, 1)
            flat = flat[::stride][: self.max_rows_per_observe]
        if self.transform_fn is not None:
            flat = np.ascontiguousarray(self.transform_fn(name, flat), dtype=np.float32)
            if flat.ndim != 2 or flat.size == 0:
                return
        flat = np.ascontiguousarray(flat, dtype=np.float32)
        cols = int(flat.shape[1])
        groups = self.sum_xxt.get(name)
        if groups is None:
            groups = []
            for start in range(0, cols, self.group_size):
                end = min(start + self.group_size, cols)
                groups.append(np.zeros((end - start, end - start), dtype=np.float64))
            self.sum_xxt[name] = groups
        for group_idx, start in enumerate(range(0, cols, self.group_size)):
            end = min(start + self.group_size, cols)
            chunk = flat[:, start:end]
            groups[group_idx] += chunk.T @ chunk
        self.counts[name] = self.counts.get(name, 0) + int(flat.shape[0])

    def finalize(self) -> dict[str, list[np.ndarray]]:
        out: dict[str, list[np.ndarray]] = {}
        for name, groups in self.sum_xxt.items():
            denom = float(max(self.counts.get(name, 0), 1))
            out[name] = [
                np.ascontiguousarray((group / denom).astype(np.float32, copy=False))
                for group in groups
            ]
        return out

class CastedLinear(nn.Module):
    def __init__(self, in_dim: int, out_dim: int, weight_name: str):
        super().__init__()
        self.weight_name = weight_name
        self.weight = nn.Linear(in_dim, out_dim, bias=False).weight.astype(mx.float32)

    def __call__(self, x: mx.array) -> mx.array:
        if _ACTIVE_ACTIVATION_COLLECTOR is not None:
            _ACTIVE_ACTIVATION_COLLECTOR.observe(self.weight_name, x)
        return x @ self.weight.astype(x.dtype).T


class RMSNormNoWeight(nn.Module):
    # MLX module wrapper around the functional RMSNorm helper so it composes nicely in blocks.
    def __call__(self, x: mx.array) -> mx.array:
        return rms_norm(x)


class CausalSelfAttention(nn.Module):
    # - separate q/k/v projections
    # - RMSNorm on q and k before attention
    # - RoPE on q and k
    # - causal masked SDPA
    def __init__(
        self,
        dim: int,
        num_heads: int,
        num_kv_heads: int,
        rope_base: float,
        qk_gain_init: float,
        prefix: str,
    ):
        super().__init__()
        if dim % num_heads != 0:
            raise ValueError("model_dim must be divisible by num_heads")
        if num_heads % num_kv_heads != 0:
            raise ValueError("num_heads must be divisible by num_kv_heads")
        self.num_heads = num_heads
        self.num_kv_heads = num_kv_heads
        self.head_dim = dim // num_heads
        if self.head_dim % 2 != 0:
            raise ValueError("head_dim must be even for RoPE")
        kv_dim = self.num_kv_heads * self.head_dim
        self.c_q = CastedLinear(dim, dim, f"{prefix}.c_q.weight")
        self.c_k = CastedLinear(dim, kv_dim, f"{prefix}.c_k.weight")
        self.c_v = CastedLinear(dim, kv_dim, f"{prefix}.c_v.weight")
        self.proj = CastedLinear(dim, dim, f"{prefix}.proj.weight")
        self.q_gain = mx.ones((num_heads,), dtype=mx.float32) * qk_gain_init
        self.rope = nn.RoPE(self.head_dim, traditional=False, base=rope_base)
        self.scale = self.head_dim ** -0.5

    def __call__(self, x: mx.array) -> mx.array:
        bsz, seqlen, dim = x.shape
        q = self.c_q(x).reshape(bsz, seqlen, self.num_heads, self.head_dim).transpose(0, 2, 1, 3)
        k = self.c_k(x).reshape(bsz, seqlen, self.num_kv_heads, self.head_dim).transpose(0, 2, 1, 3)
        v = self.c_v(x).reshape(bsz, seqlen, self.num_kv_heads, self.head_dim).transpose(0, 2, 1, 3)

        q = self.rope(rms_norm(q).astype(COMPUTE_DTYPE))
        k = self.rope(rms_norm(k).astype(COMPUTE_DTYPE))
        q = q * self.q_gain.astype(q.dtype)[None, :, None, None]
        y = mx.fast.scaled_dot_product_attention(q, k, v, scale=self.scale, mask="causal")
        y = y.transpose(0, 2, 1, 3).reshape(bsz, seqlen, dim)
        return self.proj(y)


class MLP(nn.Module):
    # Baseline MLP uses relu^2 instead of GELU/SiLU. It is cheap and works well in this setup.
    def __init__(self, dim: int, mlp_mult: int, prefix: str):
        super().__init__()
        hidden = dim * mlp_mult
        self.fc = CastedLinear(dim, hidden, f"{prefix}.fc.weight")
        self.proj = CastedLinear(hidden, dim, f"{prefix}.proj.weight")

    def __call__(self, x: mx.array) -> mx.array:
        x = nn.relu(self.fc(x))
        return self.proj(x * x)


class Block(nn.Module):
    def __init__(
        self,
        block_idx: int,
        dim: int,
        num_heads: int,
        num_kv_heads: int,
        mlp_mult: int,
        rope_base: float,
        qk_gain_init: float,
    ):
        super().__init__()
        self.attn_norm = RMSNormNoWeight()
        self.mlp_norm = RMSNormNoWeight()
        block_prefix = f"blocks.{block_idx}"
        self.attn = CausalSelfAttention(
            dim,
            num_heads,
            num_kv_heads,
            rope_base,
            qk_gain_init,
            prefix=f"{block_prefix}.attn",
        )
        self.mlp = MLP(dim, mlp_mult, prefix=f"{block_prefix}.mlp")
        self.attn_scale = mx.ones((dim,), dtype=mx.float32)
        self.mlp_scale = mx.ones((dim,), dtype=mx.float32)
        self.resid_mix = mx.array(np.stack((np.ones((dim,), dtype=np.float32), np.zeros((dim,), dtype=np.float32))))

    def __call__(self, x: mx.array, x0: mx.array) -> mx.array:
        mix = self.resid_mix.astype(x.dtype)
        x = mix[0][None, None, :] * x + mix[1][None, None, :] * x0
        attn_out = self.attn(self.attn_norm(x))
        x = x + self.attn_scale.astype(x.dtype)[None, None, :] * attn_out
        x = x + self.mlp_scale.astype(x.dtype)[None, None, :] * self.mlp(self.mlp_norm(x))
        return x


class GPT(nn.Module):
    # - token embedding + RMSNorm
    # - encoder half accumulates skip tensors
    # - decoder half consumes reversed skips with learned skip_weights
    # - tied embeddings for the LM head (the baseline default setup)
    def __init__(self, vocab_size: int, num_layers: int, dim: int, num_heads: int, num_kv_heads: int, mlp_mult: int,
                 logit_chunk_tokens: int, logit_softcap: float, rope_base: float, tied_embed_init_std: float,
                 qk_gain_init: float):
        super().__init__()
        if logit_softcap <= 0.0:
            raise ValueError(f"logit_softcap must be positive, got {logit_softcap}")
        self.logit_chunk_tokens = logit_chunk_tokens
        self.logit_softcap = logit_softcap

        self.tok_emb = nn.Embedding(vocab_size, dim)
        self.num_encoder_layers = num_layers // 2
        self.num_decoder_layers = num_layers - self.num_encoder_layers
        self.num_skip_weights = min(self.num_encoder_layers, self.num_decoder_layers)
        self.skip_weights = mx.ones((self.num_skip_weights, dim), dtype=mx.float32)
        self.blocks = [
            Block(i, dim, num_heads, num_kv_heads, mlp_mult, rope_base, qk_gain_init)
            for i in range(num_layers)
        ]
        self.final_norm = RMSNormNoWeight()

        for b in self.blocks:
            b.attn.proj.weight = mx.zeros_like(b.attn.proj.weight)
            b.mlp.proj.weight = mx.zeros_like(b.mlp.proj.weight)
        self.tok_emb.weight = (
            mx.random.normal(self.tok_emb.weight.shape, dtype=mx.float32) * tied_embed_init_std
        ).astype(COMPUTE_DTYPE)

    def softcap(self, logits: mx.array) -> mx.array:
        c = self.logit_softcap
        return c * mx.tanh(logits / c)

    def __call__(self, input_ids: mx.array) -> mx.array:
        x = rms_norm(self.tok_emb(input_ids).astype(COMPUTE_DTYPE))
        x0 = x
        skips: list[mx.array] = []

        for i in range(self.num_encoder_layers):
            x = self.blocks[i](x, x0)
            skips.append(x)
        for i in range(self.num_decoder_layers):
            # Odd layer counts have one more decoder block than encoder block. The baseline only
            # applies a skip connection when one exists, then runs the remaining decoder block(s)
            # without an added skip.
            if skips:
                x = x + self.skip_weights[i].astype(x.dtype)[None, None, :] * skips.pop()
            x = self.blocks[self.num_encoder_layers + i](x, x0)
        x = self.final_norm(x)
        if _ACTIVE_ACTIVATION_COLLECTOR is not None and QUANTIZE_TIED_HEAD:
            _ACTIVE_ACTIVATION_COLLECTOR.observe("tok_emb.weight", x)
        return x

    def loss(self, input_ids: mx.array, target_ids: mx.array) -> mx.array:
        # Cross-entropy over flattened tokens. We keep optional logit chunking because it is a useful
        # memory knob on Macs, but the common path is chunk_tokens=0 (single matmul + CE).
        x = self(input_ids).reshape(-1, self.tok_emb.weight.shape[1])
        y = target_ids.reshape(-1)
        if self.logit_chunk_tokens <= 0 or x.shape[0] <= self.logit_chunk_tokens:
            logits_proj = x @ self.tok_emb.weight.astype(x.dtype).T
            logits = self.softcap(logits_proj)
            return nn.losses.cross_entropy(logits.astype(mx.float32), y, reduction="mean")

        loss_sum = mx.array(0.0, dtype=mx.float32)
        n = int(x.shape[0])
        for s in range(0, n, self.logit_chunk_tokens):
            e = min(s + self.logit_chunk_tokens, n)
            logits_proj = x[s:e] @ self.tok_emb.weight.astype(x.dtype).T
            logits = self.softcap(logits_proj)
            loss_sum = loss_sum + nn.losses.cross_entropy(logits.astype(mx.float32), y[s:e], reduction="sum")
        return loss_sum / float(n)

class Muon:
    # Muon applies SGD-momentum to matrix gradients, then orthogonalizes the result before the
    # parameter update.
    def __init__(self, keys: list[str], params: dict[str, mx.array], args: Hyperparameters):
        self.keys = keys
        self.args = args
        self.buffers = {k: mx.zeros_like(params[k]) for k in keys}

    def step(self, params: dict[str, mx.array], grads: dict[str, mx.array], step: int, lr_mul: float) -> dict[str, mx.array]:
        if self.args.muon_momentum_warmup_steps:
            t = min(step / self.args.muon_momentum_warmup_steps, 1.0)
            momentum = (1.0 - t) * self.args.muon_momentum_warmup_start + t * self.args.muon_momentum
        else:
            momentum = self.args.muon_momentum
        lr = self.args.matrix_lr * lr_mul
        out: dict[str, mx.array] = {}
        for k in self.keys:
            p = params[k]
            g = grads[k]
            buf = momentum * self.buffers[k] + g
            self.buffers[k] = buf
            g_eff = g + momentum * buf
            g_ortho = zeropower_newtonschulz5(g_eff, self.args.muon_backend_steps)
            scale = math.sqrt(max(1.0, float(p.shape[0]) / float(p.shape[1])))
            out[k] = p - lr * (g_ortho * scale).astype(p.dtype)
        return out


class SplitOptimizers:
    # - embeddings: Adam with the tied-embedding LR
    # - block matrices (2D): Muon
    # - block scalars + skip weights: Adam
    # This preserves the high-level optimization behavior even though MLX internals differ.
    def __init__(self, model: GPT, args: Hyperparameters):
        self.args = args
        params = dict(tree_flatten(model.parameters()))
        self.embed_key = "tok_emb.weight"
        self.matrix_keys = [
            k
            for k, p in params.items()
            if k.startswith("blocks.") and p.ndim == 2 and not any(pattern in k for pattern in CONTROL_TENSOR_NAME_PATTERNS)
        ]
        self.scalar_keys = [
            k
            for k, p in params.items()
            if k == "skip_weights" or (k.startswith("blocks.") and (p.ndim < 2 or any(pattern in k for pattern in CONTROL_TENSOR_NAME_PATTERNS)))
        ]

        self.muon = Muon(self.matrix_keys, params, args)
        self.adam_embed = optim.Adam(
            learning_rate=args.tied_embed_lr,
            betas=[args.beta1, args.beta2],
            eps=args.adam_eps,
            bias_correction=True,
        )
        self.adam_scalar = optim.Adam(
            learning_rate=args.scalar_lr,
            betas=[args.beta1, args.beta2],
            eps=args.adam_eps,
            bias_correction=True,
        )

    def step(self, model: GPT, grads_tree: dict, step: int, lr_mul: float) -> None:
        params = dict(tree_flatten(model.parameters()))
        grads = dict(tree_flatten(grads_tree))
        updated = dict(params)

        updated.update(self.muon.step(params, grads, step=step, lr_mul=lr_mul))

        self.adam_embed.learning_rate = self.args.tied_embed_lr * lr_mul
        updated.update(
            self.adam_embed.apply_gradients(
                {self.embed_key: grads[self.embed_key]},
                {self.embed_key: params[self.embed_key]},
            )
        )

        self.adam_scalar.learning_rate = self.args.scalar_lr * lr_mul
        scalar_grads = {k: grads[k] for k in self.scalar_keys}
        scalar_params = {k: params[k] for k in self.scalar_keys}
        updated.update(self.adam_scalar.apply_gradients(scalar_grads, scalar_params))

        model.update(tree_unflatten(list(updated.items())))

def is_power_of_two(value: int) -> bool:
    return value > 0 and (value & (value - 1)) == 0


def hadamard_transform_last_axis(arr: np.ndarray) -> np.ndarray:
    if arr.ndim == 0:
        return np.array(arr, dtype=np.float32, copy=True)
    width = int(arr.shape[-1])
    if not is_power_of_two(width):
        raise ValueError(f"Hadamard transform requires a power-of-two width, got {width}")
    out = np.ascontiguousarray(arr.astype(np.float32, copy=True))
    view = out.reshape(-1, width)
    inv_sqrt2 = np.float32(1.0 / math.sqrt(2.0))
    step = 1
    while step < width:
        block = step * 2
        for start in range(0, width, block):
            mid = start + step
            end = start + block
            left = view[:, start:mid].copy()
            right = view[:, mid:end].copy()
            view[:, start:mid] = (left + right) * inv_sqrt2
            view[:, mid:end] = (left - right) * inv_sqrt2
        step = block
    return out


def deterministic_sign_vector(name: str, width: int) -> np.ndarray:
    seed_material = f"{ROTATION_SEED}:{name}:{width}".encode("utf-8")
    seed = int.from_bytes(hashlib.blake2b(seed_material, digest_size=8).digest(), "little", signed=False)
    rng = np.random.default_rng(seed)
    return np.ascontiguousarray(rng.choice(np.array((-1.0, 1.0), dtype=np.float32), size=width).astype(np.float32))


def maybe_make_rotation_signs(name: str, arr: np.ndarray) -> np.ndarray | None:
    if INPUT_ROTATION != "hadamard" or arr.ndim != 2 or not is_power_of_two(int(arr.shape[1])):
        return None
    return deterministic_sign_vector(name, int(arr.shape[1]))


def apply_input_rotation(arr: np.ndarray, sign: np.ndarray | None) -> np.ndarray:
    if sign is None:
        return np.ascontiguousarray(arr.astype(np.float32, copy=False))
    return hadamard_transform_last_axis(arr.astype(np.float32, copy=False) * sign[None, :])


def undo_input_rotation(arr: np.ndarray, sign: np.ndarray | None) -> np.ndarray:
    if sign is None:
        return np.ascontiguousarray(arr.astype(np.float32, copy=False))
    return np.ascontiguousarray(hadamard_transform_last_axis(arr) * sign[None, :], dtype=np.float32)


def quant_source_input_transform(name: str, flat: np.ndarray) -> np.ndarray:
    sign = maybe_make_rotation_signs(name, flat)
    return undo_input_rotation(flat, sign)


_SCALAR_LEVEL_CACHE: dict[tuple[str, int], np.ndarray] = {}


def scalar_quantizer_levels(qmax: int, scalar_quantizer: str | None = None) -> np.ndarray:
    scalar_quantizer = SCALAR_QUANTIZER if scalar_quantizer is None else scalar_quantizer
    key = (scalar_quantizer, int(qmax))
    cached = _SCALAR_LEVEL_CACHE.get(key)
    if cached is not None:
        return cached
    if scalar_quantizer == "uniform":
        levels = np.arange(-qmax, qmax + 1, dtype=np.float32)
    elif scalar_quantizer == "normal_quantile":
        normal = NormalDist()
        num_levels = 2 * qmax + 1
        levels = np.array(
            [normal.inv_cdf((idx + 0.5) / num_levels) for idx in range(num_levels)],
            dtype=np.float32,
        )
        levels[qmax] = 0.0
    elif scalar_quantizer == "nf4":
        levels = np.array(
            (
                -1.0,
                -0.6961928009986877,
                -0.5250730514526367,
                -0.39491748809814453,
                -0.28444138169288635,
                -0.18477343022823334,
                -0.09105003625154495,
                0.0,
                0.07958029955625534,
                0.16093020141124725,
                0.24611230194568634,
                0.33791524171829224,
                0.44070982933044434,
                0.5626170039176941,
                0.7229568362236023,
                1.0,
            ),
            dtype=np.float32,
        )
    elif scalar_quantizer == "fp4":
        # Exact bitsandbytes FP4 codebook after normalization to [-1, 1].
        # We store it in sorted value order because our scalar quantizer maps
        # via searchsorted boundaries rather than bit-pattern order.
        levels = np.array(
            (
                -1.0,
                -2.0 / 3.0,
                -0.5,
                -1.0 / 3.0,
                -0.25,
                -1.0 / 6.0,
                -1.0 / 192.0,
                0.0,
                0.0,
                1.0 / 192.0,
                1.0 / 6.0,
                0.25,
                1.0 / 3.0,
                0.5,
                2.0 / 3.0,
                1.0,
            ),
            dtype=np.float32,
        )
    elif scalar_quantizer == "af4":
        # Exact AF4 codebook values for blocksize 64 from bitsandbytes.
        # As with FP4, we keep the values in sorted order for searchsorted.
        levels = np.array(
            (
                -1.0,
                -0.69441008,
                -0.51243739,
                -0.3736951,
                -0.25607552,
                -0.14982478,
                -0.04934812,
                0.0,
                0.04273164,
                0.12934483,
                0.21961274,
                0.31675666,
                0.42563882,
                0.55496234,
                0.72424863,
                1.0,
            ),
            dtype=np.float32,
        )
    else:
        raise ValueError(f"Unsupported scalar quantizer {scalar_quantizer}")
    levels = np.ascontiguousarray(levels.astype(np.float32, copy=False))
    _SCALAR_LEVEL_CACHE[key] = levels
    return levels


def scalar_quantizer_peak_level(qmax: int, scalar_quantizer: str | None = None) -> float:
    levels = scalar_quantizer_levels(qmax, scalar_quantizer=scalar_quantizer)
    return max(float(np.max(np.abs(levels))), 1.0)


def quantize_with_fixed_row_scale(
    values: np.ndarray,
    scale: np.ndarray,
    qmax: int,
    scalar_quantizer: str | None = None,
    round_bias: float | np.ndarray | None = None,
) -> tuple[np.ndarray, np.ndarray]:
    scalar_quantizer = SCALAR_QUANTIZER if scalar_quantizer is None else scalar_quantizer
    bias: float | np.ndarray
    if round_bias is None:
        bias = 0.0
    else:
        bias_arr = np.asarray(round_bias, dtype=np.float32)
        if bias_arr.ndim == 0:
            bias = float(bias_arr)
        else:
            bias = np.ascontiguousarray(bias_arr.reshape(-1, 1), dtype=np.float32)
    if scalar_quantizer == "uniform":
        q = np.clip(np.round(values / scale[:, None] + bias), -qmax, qmax).astype(np.int8, copy=False)
        dq = q.astype(np.float32) * scale[:, None]
        return np.ascontiguousarray(q), np.ascontiguousarray(dq.astype(np.float32, copy=False))

    levels = scalar_quantizer_levels(qmax, scalar_quantizer=scalar_quantizer)
    boundaries = 0.5 * (levels[:-1] + levels[1:])
    clipped = np.clip(values / scale[:, None] + bias, float(levels[0]), float(levels[-1]))
    indices = np.searchsorted(boundaries, clipped, side="right")
    if scalar_quantizer in {"nf4", "fp4", "af4"}:
        q = np.clip(indices, 0, len(levels) - 1).astype(np.int8, copy=False)
        dq = levels[np.clip(q.astype(np.int16), 0, len(levels) - 1)] * scale[:, None]
        return np.ascontiguousarray(q), np.ascontiguousarray(dq.astype(np.float32, copy=False))
    q = np.clip(indices - qmax, -qmax, qmax).astype(np.int8, copy=False)
    dq = levels[np.clip(q.astype(np.int16) + qmax, 0, 2 * qmax)] * scale[:, None]
    return np.ascontiguousarray(q), np.ascontiguousarray(dq.astype(np.float32, copy=False))


def dequantize_codes(
    q_np: np.ndarray,
    scale: np.ndarray,
    qmax: int,
    scalar_quantizer: str | None = None,
) -> np.ndarray:
    scalar_quantizer = SCALAR_QUANTIZER if scalar_quantizer is None else scalar_quantizer
    if scalar_quantizer == "uniform":
        return q_np.astype(np.float32) * scale
    levels = scalar_quantizer_levels(qmax, scalar_quantizer=scalar_quantizer)
    if scalar_quantizer in {"nf4", "fp4", "af4"}:
        indices = np.clip(q_np.astype(np.int16), 0, len(levels) - 1)
    else:
        indices = np.clip(q_np.astype(np.int16) + qmax, 0, 2 * qmax)
    return levels[indices] * scale


def select_sparse_residual_entries(residual: np.ndarray) -> tuple[np.ndarray | None, np.ndarray | None]:
    if SPARSE_RESIDUAL_FRAC <= 0.0:
        return None, None
    flat = np.ascontiguousarray(residual.reshape(-1), dtype=np.float32)
    if flat.size == 0:
        return None, None
    k = int(round(SPARSE_RESIDUAL_FRAC * flat.size))
    if k <= 0:
        return None, None
    k = min(k, flat.size)
    scores = np.abs(flat)
    if k == flat.size:
        idx = np.arange(flat.size, dtype=np.uint32)
    else:
        idx = np.argpartition(scores, -k)[-k:].astype(np.uint32, copy=False)
        idx.sort()
    values = np.ascontiguousarray(flat[idx].astype(SPARSE_RESIDUAL_STORE_DTYPE, copy=False))
    return np.ascontiguousarray(idx), values


def add_sparse_residual(out_arr: np.ndarray, indices: np.ndarray | None, values: np.ndarray | None) -> np.ndarray:
    if indices is None or values is None or indices.size == 0:
        return out_arr
    flat = out_arr.reshape(-1)
    flat[np.asarray(indices, dtype=np.int64)] += np.asarray(values, dtype=np.float32)
    return out_arr


def zero_flat_indices(arr: np.ndarray, indices: np.ndarray | None) -> np.ndarray:
    if indices is None or indices.size == 0:
        return np.ascontiguousarray(arr.astype(np.float32, copy=False))
    out = np.ascontiguousarray(arr.astype(np.float32, copy=True))
    out.reshape(-1)[np.asarray(indices, dtype=np.int64)] = 0.0
    return out


def quantize_float32_array(f32: np.ndarray, qmax: int) -> tuple[np.ndarray, np.ndarray]:
    peak_level = scalar_quantizer_peak_level(qmax)
    if f32.ndim == 2:
        # Matrices use per-row group scales so one local outlier chunk does not
        # set the scale for an entire row at low bitwidth.
        rows, cols = f32.shape
        if cols > INT8_PER_ROW_GROUP_SIZE:
            num_groups = math.ceil(cols / INT8_PER_ROW_GROUP_SIZE)
            q = np.empty((rows, cols), dtype=np.int8)
            scale = np.empty((rows, num_groups), dtype=np.float32)
            for group_idx in range(num_groups):
                start = group_idx * INT8_PER_ROW_GROUP_SIZE
                end = min(start + INT8_PER_ROW_GROUP_SIZE, cols)
                chunk = f32[:, start:end]
                clip_abs = (
                    np.quantile(np.abs(chunk), INT8_CLIP_Q, axis=1)
                    if chunk.size
                    else np.empty((rows,), dtype=np.float32)
                )
                group_scale = np.maximum(
                    clip_abs / peak_level,
                    1.0 / peak_level,
                ).astype(np.float32, copy=False)
                q_chunk, _ = quantize_with_fixed_row_scale(chunk, group_scale, qmax=qmax)
                q[:, start:end] = q_chunk
                scale[:, group_idx] = group_scale
            return np.ascontiguousarray(q), np.ascontiguousarray(scale.astype(INT8_PER_ROW_SCALE_DTYPE, copy=False))

        clip_abs = np.quantile(np.abs(f32), INT8_CLIP_Q, axis=1) if f32.size else np.empty((rows,), dtype=np.float32)
        scale = np.maximum(clip_abs / peak_level, 1.0 / peak_level).astype(np.float32, copy=False)
        q, _ = quantize_with_fixed_row_scale(f32, scale, qmax=qmax)
        return np.ascontiguousarray(q), np.ascontiguousarray(scale.astype(INT8_PER_ROW_SCALE_DTYPE, copy=False))

    # Vectors / scalars use a simpler per-tensor scale.
    clip_abs = float(np.quantile(np.abs(f32).reshape(-1), INT8_CLIP_Q)) if f32.size else 0.0
    scale = np.array(clip_abs / peak_level if clip_abs > 0.0 else 1.0, dtype=np.float32)
    q, _ = quantize_with_fixed_row_scale(
        np.ascontiguousarray(f32.reshape(1, -1)),
        np.array((float(scale),), dtype=np.float32),
        qmax=qmax,
    )
    q = q.reshape(f32.shape)
    return np.ascontiguousarray(q), scale


def scale_hessian_groups_for_column_scale(
    hessian_groups: list[np.ndarray] | None,
    column_scale: np.ndarray | None,
) -> list[np.ndarray] | None:
    if hessian_groups is None or column_scale is None:
        return hessian_groups
    scaled_groups: list[np.ndarray] = []
    for group_idx, hessian in enumerate(hessian_groups):
        hessian_f32 = np.asarray(hessian, dtype=np.float32)
        start = group_idx * INT8_PER_ROW_GROUP_SIZE
        end = start + hessian_f32.shape[0]
        scale_chunk = np.asarray(column_scale[start:end], dtype=np.float32)
        if scale_chunk.shape[0] != hessian_f32.shape[0]:
            scaled_groups.append(hessian_f32)
            continue
        scaled_groups.append(
            np.ascontiguousarray(
                hessian_f32 / scale_chunk[None, :] / scale_chunk[:, None],
                dtype=np.float32,
            )
        )
    return scaled_groups


def reconstruction_error_rows(
    diff: np.ndarray,
    hessian: np.ndarray | None,
) -> np.ndarray:
    diff64 = np.asarray(diff, dtype=np.float64)
    if hessian is None:
        return np.sum(diff64 * diff64, axis=1, dtype=np.float64)
    hessian64 = np.asarray(hessian, dtype=np.float64)
    if ROUND_OPT_USE_FULL_HESSIAN and hessian64.ndim == 2 and hessian64.shape[0] == hessian64.shape[1]:
        return np.einsum("rc,cd,rd->r", diff64, hessian64, diff64, optimize=True)
    diag = np.diag(hessian64) if hessian64.ndim == 2 else hessian64.reshape(-1)
    return np.sum(diff64 * diff64 * diag[None, :], axis=1, dtype=np.float64)


def signroundlite_quantize_group_matrix(
    group: np.ndarray,
    qmax: int,
    hessian: np.ndarray | None,
) -> tuple[np.ndarray, np.ndarray]:
    rows, cols = group.shape
    if hessian is not None and hessian.shape != (cols, cols):
        hessian = None
    clip_abs = np.quantile(np.abs(group), INT8_CLIP_Q, axis=1) if group.size else np.empty((rows,), dtype=np.float32)
    peak_level = scalar_quantizer_peak_level(qmax)
    base_scale = np.maximum(clip_abs / peak_level, 1.0 / peak_level).astype(np.float32, copy=False)
    base_q, base_dq = quantize_with_fixed_row_scale(group, base_scale, qmax=qmax)
    base_err = reconstruction_error_rows(group - base_dq, hessian)

    if not ROUND_OPT_ROWWISE_SELECTION:
        best_q = base_q
        best_scale = base_scale
        best_mean_error = float(np.mean(base_err))
        for scale_mult in ROUND_OPT_SCALE_GRID:
            candidate_scale = np.maximum(base_scale * float(scale_mult), 1.0 / peak_level).astype(np.float32, copy=False)
            for bias in ROUND_OPT_BIAS_GRID:
                if abs(scale_mult - 1.0) <= 1e-12 and abs(bias) <= 1e-12:
                    continue
                candidate_q, candidate_dq = quantize_with_fixed_row_scale(
                    group,
                    candidate_scale,
                    qmax=qmax,
                    round_bias=float(bias),
                )
                candidate_err = reconstruction_error_rows(group - candidate_dq, hessian)
                candidate_mean_error = float(np.mean(candidate_err))
                if candidate_mean_error + 1e-12 < best_mean_error:
                    best_mean_error = candidate_mean_error
                    best_q = candidate_q
                    best_scale = candidate_scale
        return np.ascontiguousarray(best_q), np.ascontiguousarray(best_scale.astype(INT8_PER_ROW_SCALE_DTYPE, copy=False))

    best_q = np.ascontiguousarray(base_q.copy())
    best_scale = np.ascontiguousarray(base_scale.copy())
    best_err = np.ascontiguousarray(base_err.copy())
    for scale_mult in ROUND_OPT_SCALE_GRID:
        candidate_scale = np.maximum(base_scale * float(scale_mult), 1.0 / peak_level).astype(np.float32, copy=False)
        for bias in ROUND_OPT_BIAS_GRID:
            if abs(scale_mult - 1.0) <= 1e-12 and abs(bias) <= 1e-12:
                continue
            candidate_q, candidate_dq = quantize_with_fixed_row_scale(
                group,
                candidate_scale,
                qmax=qmax,
                round_bias=float(bias),
            )
            candidate_err = reconstruction_error_rows(group - candidate_dq, hessian)
            better = candidate_err + 1e-12 < best_err
            if np.any(better):
                best_err[better] = candidate_err[better]
                best_q[better] = candidate_q[better]
                best_scale[better] = candidate_scale[better]
    return np.ascontiguousarray(best_q), np.ascontiguousarray(best_scale.astype(INT8_PER_ROW_SCALE_DTYPE, copy=False))


def regularized_hessian_inverse(hessian: np.ndarray) -> np.ndarray:
    h = np.asarray(hessian, dtype=np.float64)
    if h.ndim != 2 or h.shape[0] != h.shape[1]:
        raise ValueError(f"Expected square Hessian, got shape {h.shape}")
    diag_mean = float(np.mean(np.diag(h))) if h.size else 1.0
    damp = max(GPTQ_DAMP_PERCENT * max(diag_mean, 0.0), 1e-4)
    h_reg = h + np.eye(h.shape[0], dtype=np.float64) * damp
    try:
        h_inv = np.linalg.inv(h_reg)
    except np.linalg.LinAlgError:
        h_inv = np.linalg.pinv(h_reg, hermitian=True)
    return np.ascontiguousarray(h_inv.astype(np.float32, copy=False))


def gptq_quantize_group_matrix(
    group: np.ndarray,
    qmax: int,
    hessian: np.ndarray | None,
) -> tuple[np.ndarray, np.ndarray]:
    rows, cols = group.shape
    clip_abs = np.quantile(np.abs(group), INT8_CLIP_Q, axis=1) if group.size else np.empty((rows,), dtype=np.float32)
    peak_level = scalar_quantizer_peak_level(qmax)
    scale = np.maximum(clip_abs / peak_level, 1.0 / peak_level).astype(np.float32, copy=False)
    if hessian is None or hessian.shape != (cols, cols):
        hessian = np.eye(cols, dtype=np.float32)
    h_inv = regularized_hessian_inverse(hessian)
    if GPTQ_ACT_ORDER and cols > 1:
        order = np.argsort(-np.diag(np.asarray(hessian, dtype=np.float32)))
    else:
        order = np.arange(cols)
    work = np.ascontiguousarray(group[:, order].astype(np.float32, copy=True))
    perm_h_inv = h_inv[np.ix_(order, order)]
    q_perm = np.empty((rows, cols), dtype=np.int8)
    for col in range(cols):
        q_col, dq_col = quantize_with_fixed_row_scale(work[:, col:col + 1], scale, qmax=qmax)
        q_perm[:, col] = q_col[:, 0]
        denom = max(float(perm_h_inv[col, col]), 1e-6)
        error = (work[:, col] - dq_col[:, 0]) / denom
        if col + 1 < cols:
            work[:, col + 1:] -= error[:, None] * perm_h_inv[col, col + 1:][None, :]
    q_out = np.empty_like(q_perm)
    q_out[:, order] = q_perm
    return np.ascontiguousarray(q_out), np.ascontiguousarray(scale.astype(INT8_PER_ROW_SCALE_DTYPE, copy=False))


def quantize_float32_array_gptq(
    f32: np.ndarray,
    qmax: int,
    hessian_groups: list[np.ndarray] | None,
) -> tuple[np.ndarray, np.ndarray]:
    if f32.ndim != 2:
        return quantize_float32_array(f32, qmax=qmax)
    rows, cols = f32.shape
    if cols > INT8_PER_ROW_GROUP_SIZE:
        num_groups = math.ceil(cols / INT8_PER_ROW_GROUP_SIZE)
        q = np.empty((rows, cols), dtype=np.int8)
        scale = np.empty((rows, num_groups), dtype=np.float32)
        for group_idx in range(num_groups):
            start = group_idx * INT8_PER_ROW_GROUP_SIZE
            end = min(start + INT8_PER_ROW_GROUP_SIZE, cols)
            chunk = f32[:, start:end]
            hessian = None
            if hessian_groups is not None and group_idx < len(hessian_groups):
                hessian = hessian_groups[group_idx]
            q_chunk, scale_chunk = gptq_quantize_group_matrix(chunk, qmax=qmax, hessian=hessian)
            q[:, start:end] = q_chunk
            scale[:, group_idx] = scale_chunk
        return np.ascontiguousarray(q), np.ascontiguousarray(scale.astype(INT8_PER_ROW_SCALE_DTYPE, copy=False))

    base_hessian = None if not hessian_groups else hessian_groups[0]
    return gptq_quantize_group_matrix(f32, qmax=qmax, hessian=base_hessian)

def dequantize_quantized_array(
    q_np: np.ndarray,
    scale: np.ndarray,
    meta: dict[str, object],
    column_scale: np.ndarray | None = None,
) -> np.ndarray:
    qmax = int(meta.get("qmax", INT8_RTN_QMAX))
    scalar_quantizer = str(meta.get("scalar_quantizer", "uniform"))
    if meta.get("scheme") == "per_row_group":
        group_size = int(meta["group_size"])
        out_arr = np.empty(q_np.shape, dtype=np.float32)
        for group_idx in range(scale.shape[1]):
            start = group_idx * group_size
            end = min(start + group_size, q_np.shape[1])
            out_arr[:, start:end] = dequantize_codes(
                q_np[:, start:end],
                scale[:, group_idx][:, None],
                qmax=qmax,
                scalar_quantizer=scalar_quantizer,
            )
    elif meta.get("scheme") == "per_row" or scale.ndim > 0:
        out_arr = dequantize_codes(
            q_np,
            scale.reshape((q_np.shape[0],) + (1,) * (q_np.ndim - 1)),
            qmax=qmax,
            scalar_quantizer=scalar_quantizer,
        )
    else:
        out_arr = dequantize_codes(
            q_np,
            np.array(float(scale), dtype=np.float32),
            qmax=qmax,
            scalar_quantizer=scalar_quantizer,
        )
    if column_scale is not None and out_arr.ndim == 2:
        out_arr = out_arr / column_scale[None, :]
    return out_arr


def make_awq_column_scale(act_abs_mean: np.ndarray, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return np.ones_like(act_abs_mean, dtype=np.float32)
    scale = np.asarray(np.maximum(act_abs_mean, AWQ_MIN_ACT_SCALE) ** alpha, dtype=np.float32)
    geom_mean = float(np.exp(np.mean(np.log(np.maximum(scale, AWQ_MIN_ACT_SCALE)))))
    scale = scale / max(geom_mean, AWQ_MIN_ACT_SCALE)
    return np.ascontiguousarray(np.clip(scale, 1.0 / AWQ_MAX_SCALE, AWQ_MAX_SCALE).astype(np.float32, copy=False))


def select_lowrank_residual_factors(
    residual: np.ndarray,
    act_abs_mean: np.ndarray | None = None,
) -> tuple[np.ndarray | None, np.ndarray | None, bool]:
    if not lowrank_residual_enabled() or residual.ndim != 2 or residual.size == 0:
        return None, None, False
    rank = min(LOWRANK_RESIDUAL_RANK, residual.shape[0], residual.shape[1])
    if rank <= 0:
        return None, None, False

    work = np.ascontiguousarray(residual.astype(np.float32, copy=False))
    used_actscale = False
    if (
        LOWRANK_RESIDUAL_MODE == "actscale"
        and act_abs_mean is not None
        and act_abs_mean.ndim == 1
        and act_abs_mean.shape[0] == residual.shape[1]
    ):
        column_scale = make_awq_column_scale(np.asarray(act_abs_mean, dtype=np.float32), alpha=1.0)
        work = np.ascontiguousarray(work * column_scale[None, :], dtype=np.float32)
        used_actscale = True
    else:
        column_scale = None

    try:
        u, s, vt = np.linalg.svd(work, full_matrices=False)
    except np.linalg.LinAlgError:
        return None, None, used_actscale
    if s.size == 0 or not np.isfinite(s[0]) or float(s[0]) <= 0.0:
        return None, None, used_actscale

    left = np.ascontiguousarray((u[:, :rank] * s[:rank][None, :]).astype(LOWRANK_RESIDUAL_STORE_DTYPE, copy=False))
    right = np.asarray(vt[:rank, :], dtype=np.float32)
    if column_scale is not None:
        right = right / column_scale[None, :]
    right = np.ascontiguousarray(right.astype(LOWRANK_RESIDUAL_STORE_DTYPE, copy=False))
    return left, right, used_actscale


def add_lowrank_residual(
    out_arr: np.ndarray,
    left: np.ndarray | None,
    right: np.ndarray | None,
) -> np.ndarray:
    if left is None or right is None or out_arr.ndim != 2:
        return out_arr
    return np.ascontiguousarray(
        out_arr.astype(np.float32, copy=False)
        + np.asarray(left, dtype=np.float32) @ np.asarray(right, dtype=np.float32),
        dtype=np.float32,
    )


def search_awq_column_scale(
    name: str,
    arr: mx.array,
    act_abs_mean: np.ndarray | None,
    qmax: int,
    log_fn: Callable[[str], None] | None = None,
) -> tuple[np.ndarray | None, float | None]:
    if not awq_name_enabled(name) or act_abs_mean is None or arr.ndim != 2:
        return None, None
    f32 = _np_float32(arr)
    importance = np.asarray(np.maximum(act_abs_mean, AWQ_MIN_ACT_SCALE), dtype=np.float32)
    importance = importance / max(float(np.mean(importance)), AWQ_MIN_ACT_SCALE)
    best_alpha = 0.0
    best_error = float("inf")
    best_column_scale = np.ones_like(importance, dtype=np.float32)
    for alpha in AWQ_ALPHA_GRID or (0.0,):
        column_scale = make_awq_column_scale(importance, alpha)
        q, s = quantize_float32_array(f32 * column_scale[None, :], qmax=qmax)
        meta = {"scheme": "per_row_group", "group_size": INT8_PER_ROW_GROUP_SIZE} if s.ndim == 2 else {"scheme": "per_row"}
        dq = dequantize_quantized_array(q, np.asarray(s, dtype=np.float32), meta, column_scale=column_scale)
        error = float(np.mean(np.square((f32 - dq) * importance[None, :]), dtype=np.float64))
        if error < best_error:
            best_error = error
            best_alpha = float(alpha)
            best_column_scale = column_scale
    if best_alpha <= 0.0:
        return None, None
    if log_fn is not None:
        log_fn(
            f"awq_tensor:{name} alpha:{best_alpha:.2f} "
            f"scale_min:{float(best_column_scale.min()):.4f} scale_max:{float(best_column_scale.max()):.4f} "
            f"weighted_mse:{best_error:.8e}"
        )
    return best_column_scale, best_alpha


def quantize_float_array(
    f32: np.ndarray,
    qmax: int,
    column_scale: np.ndarray | None = None,
    gptq_hessian_groups: list[np.ndarray] | None = None,
    roundopt_hessian_groups: list[np.ndarray] | None = None,
) -> tuple[np.ndarray, np.ndarray]:
    if column_scale is not None and f32.ndim == 2:
        f32 = f32 * column_scale[None, :]
    if GPTQ_ENABLED:
        return quantize_float32_array_gptq(f32, qmax=qmax, hessian_groups=gptq_hessian_groups)
    if round_opt_enabled():
        return quantize_float32_array_signroundlite(f32, qmax=qmax, hessian_groups=roundopt_hessian_groups)
    return quantize_float32_array(f32, qmax=qmax)


def quantize_state_dict_int8(
    flat_state: dict[str, mx.array],
    awq_activation_scales: dict[str, np.ndarray] | None = None,
    gptq_hessian_groups: dict[str, list[np.ndarray]] | None = None,
    roundopt_hessian_groups: dict[str, list[np.ndarray]] | None = None,
    log_fn: Callable[[str], None] | None = None,
) -> tuple[dict[str, object], dict[str, int]]:
    quantized: dict[str, np.ndarray] = {}
    scales: dict[str, np.ndarray] = {}
    awq_scales: dict[str, np.ndarray] = {}
    rotation_signs: dict[str, np.ndarray] = {}
    sparse_residual_indices: dict[str, np.ndarray] = {}
    sparse_residual_values: dict[str, np.ndarray] = {}
    lowrank_residual_left: dict[str, np.ndarray] = {}
    lowrank_residual_right: dict[str, np.ndarray] = {}
    dtypes: dict[str, str] = {}
    passthrough: dict[str, np.ndarray] = {}
    passthrough_orig_dtypes: dict[str, str] = {}
    qmeta: dict[str, dict[str, object]] = {}
    stats = dict.fromkeys(
        (
            "param_count",
            "num_tensors",
            "num_float_tensors",
            "num_nonfloat_tensors",
            "baseline_tensor_bytes",
            "int8_payload_bytes",
            "num_awq_tensors",
            "awq_scale_bytes",
            "num_gptq_tensors",
            "num_roundopt_tensors",
            "num_rotated_tensors",
            "rotation_bytes",
            "num_sparse_residual_tensors",
            "sparse_residual_values",
            "sparse_residual_bytes",
            "num_lowrank_residual_tensors",
            "lowrank_residual_rank_sum",
            "lowrank_residual_bytes",
        ),
        0,
    )
    for name, arr in flat_state.items():
        stats["param_count"] += int(arr.size)
        stats["num_tensors"] += 1
        stats["baseline_tensor_bytes"] += int(arr.nbytes)
        if not mx.issubdtype(arr.dtype, mx.floating):
            stats["num_nonfloat_tensors"] += 1
            passthrough[name] = np.ascontiguousarray(np.array(arr))
            stats["int8_payload_bytes"] += int(passthrough[name].nbytes)
            continue

        # Small float tensors are cheap enough to keep directly. We still downcast
        # fp32/bf16 passthrough tensors to fp16 so metadata does not dominate size.
        if int(arr.size) <= INT8_KEEP_FLOAT_MAX_NUMEL or (name == "tok_emb.weight" and not QUANTIZE_TIED_HEAD):
            kept = keep_float_array(name, arr, passthrough_orig_dtypes)
            passthrough[name] = kept
            stats["int8_payload_bytes"] += int(kept.nbytes)
            continue

        stats["num_float_tensors"] += 1
        qmax = INT8_HEAD_QMAX if name == "tok_emb.weight" else INT8_RTN_QMAX
        f32 = _np_float32(arr)
        sign = maybe_make_rotation_signs(name, f32)
        quant_source = apply_input_rotation(f32, sign)
        awq_arr = mx.array(quant_source, dtype=arr.dtype) if AWQ_ENABLED else arr
        column_scale, awq_alpha = search_awq_column_scale(
            name,
            awq_arr,
            None if awq_activation_scales is None else awq_activation_scales.get(name),
            qmax=qmax,
            log_fn=log_fn,
        )
        gptq_groups = None if gptq_hessian_groups is None else gptq_hessian_groups.get(name)
        roundopt_groups = None if roundopt_hessian_groups is None else roundopt_hessian_groups.get(name)
        if column_scale is not None:
            gptq_groups = scale_hessian_groups_for_column_scale(gptq_groups, column_scale)
            roundopt_groups = scale_hessian_groups_for_column_scale(roundopt_groups, column_scale)
        q, s = quantize_float_array(
            quant_source,
            qmax=qmax,
            column_scale=column_scale,
            gptq_hessian_groups=gptq_groups,
            roundopt_hessian_groups=roundopt_groups,
        )
        meta: dict[str, object] = {"qmax": int(qmax), "scalar_quantizer": SCALAR_QUANTIZER}
        if s.ndim == 2:
            meta.update({"scheme": "per_row_group", "axis": 0, "group_size": INT8_PER_ROW_GROUP_SIZE})
        elif s.ndim > 0:
            meta.update({"scheme": "per_row", "axis": 0})
        if column_scale is not None:
            meta["awq_alpha"] = float(awq_alpha)
            awq_scale_arr = np.ascontiguousarray(column_scale.astype(AWQ_SCALE_STORE_DTYPE, copy=False))
            awq_scales[name] = awq_scale_arr
            stats["int8_payload_bytes"] += int(awq_scale_arr.nbytes)
            stats["num_awq_tensors"] += 1
            stats["awq_scale_bytes"] += int(awq_scale_arr.nbytes)
        if sign is not None:
            rotation_arr = np.ascontiguousarray(sign.astype(np.int8, copy=False))
            rotation_signs[name] = rotation_arr
            meta["input_rotation"] = INPUT_ROTATION
            stats["num_rotated_tensors"] += 1
            stats["rotation_bytes"] += int(rotation_arr.nbytes)
            stats["int8_payload_bytes"] += int(rotation_arr.nbytes)
        if gptq_hessian_groups is not None and name in gptq_hessian_groups:
            meta["quant_method"] = QUANT_METHOD
            stats["num_gptq_tensors"] += 1
        if roundopt_hessian_groups is not None and name in roundopt_hessian_groups:
            meta["round_opt_method"] = ROUND_OPT_METHOD
            stats["num_roundopt_tensors"] += 1
        dense_rot = dequantize_quantized_array(
            q,
            np.asarray(s, dtype=np.float32),
            meta,
            column_scale=column_scale,
        )
        dense_orig = undo_input_rotation(dense_rot, sign)
        residual = np.ascontiguousarray(f32 - dense_orig, dtype=np.float32)
        sparse_idx, sparse_vals = select_sparse_residual_entries(residual)
        if sparse_idx is not None and sparse_vals is not None and sparse_idx.size > 0:
            sparse_residual_indices[name] = sparse_idx
            sparse_residual_values[name] = sparse_vals
            meta["sparse_residual_frac"] = float(SPARSE_RESIDUAL_FRAC)
            meta["sparse_residual_mode"] = SPARSE_RESIDUAL_MODE
            stats["num_sparse_residual_tensors"] += 1
            stats["sparse_residual_values"] += int(sparse_idx.size)
            sparse_bytes = int(sparse_idx.nbytes + sparse_vals.nbytes)
            stats["sparse_residual_bytes"] += sparse_bytes
            stats["int8_payload_bytes"] += sparse_bytes
        lowrank_left, lowrank_right, lowrank_used_actscale = select_lowrank_residual_factors(
            zero_flat_indices(residual, sparse_idx),
            None if awq_activation_scales is None else awq_activation_scales.get(name),
        )
        if lowrank_left is not None and lowrank_right is not None:
            lowrank_residual_left[name] = lowrank_left
            lowrank_residual_right[name] = lowrank_right
            meta["lowrank_residual_rank"] = int(lowrank_left.shape[1])
            meta["lowrank_residual_mode"] = LOWRANK_RESIDUAL_MODE
            meta["lowrank_residual_actscaled"] = int(lowrank_used_actscale)
            stats["num_lowrank_residual_tensors"] += 1
            stats["lowrank_residual_rank_sum"] += int(lowrank_left.shape[1])
            lowrank_bytes = int(lowrank_left.nbytes + lowrank_right.nbytes)
            stats["lowrank_residual_bytes"] += lowrank_bytes
            stats["int8_payload_bytes"] += lowrank_bytes
        if meta:
            qmeta[name] = meta
        quantized[name] = q
        scales[name] = s
        dtypes[name] = str(arr.dtype).split(".")[-1]
        stats["int8_payload_bytes"] += int(q.nbytes + s.nbytes)
    body_tag = f"int{signed_qmax_to_bits(INT8_RTN_QMAX)}_group{INT8_PER_ROW_GROUP_SIZE}"
    head_tag = f"headint{signed_qmax_to_bits(INT8_HEAD_QMAX)}" if QUANTIZE_TIED_HEAD else "tiedemb_fp16"
    awq_tag = "_awq" if AWQ_ENABLED else ""
    method_tag = f"_{QUANT_METHOD}" if QUANT_METHOD != "rtn" else ""
    roundopt_tag = f"_{ROUND_OPT_METHOD}" if round_opt_enabled() else ""
    rotation_tag = "_rht" if INPUT_ROTATION == "hadamard" else ""
    scalar_tag = (
        "_normalq"
        if SCALAR_QUANTIZER == "normal_quantile"
        else (
            "_nf4"
            if SCALAR_QUANTIZER == "nf4"
            else ("_fp4" if SCALAR_QUANTIZER == "fp4" else ("_af4" if SCALAR_QUANTIZER == "af4" else ""))
        )
    )
    sparse_tag = "_sparseresid" if SPARSE_RESIDUAL_FRAC > 0.0 else ""
    lowrank_tag = (
        f"_lowrankr{LOWRANK_RESIDUAL_RANK}{'_act' if LOWRANK_RESIDUAL_MODE == 'actscale' else ''}"
        if lowrank_residual_enabled()
        else ""
    )
    obj: dict[str, object] = {
        "__quant_format__": (
            f"{body_tag}_{head_tag}{rotation_tag}{method_tag}{roundopt_tag}{scalar_tag}{sparse_tag}{lowrank_tag}{awq_tag}_per_row_v1"
        ),
        "quantized": quantized,
        "scales": scales,
        "dtypes": dtypes,
        "passthrough": passthrough,
    }
    if qmeta:
        obj["qmeta"] = qmeta
    if awq_scales:
        obj["awq_scales"] = awq_scales
    if rotation_signs:
        obj["rotation_signs"] = rotation_signs
    if sparse_residual_indices:
        obj["sparse_residual_indices"] = sparse_residual_indices
        obj["sparse_residual_values"] = sparse_residual_values
    if lowrank_residual_left:
        obj["lowrank_residual_left"] = lowrank_residual_left
        obj["lowrank_residual_right"] = lowrank_residual_right
    if passthrough_orig_dtypes:
        obj["passthrough_orig_dtypes"] = passthrough_orig_dtypes
    return obj, stats


def dequantize_state_dict_int8(quant_obj: dict[str, object]) -> dict[str, mx.array]:
    out: dict[str, mx.array] = {}
    qmeta = quant_obj.get("qmeta", {})
    awq_scales = quant_obj.get("awq_scales", {})
    rotation_signs = quant_obj.get("rotation_signs", {})
    sparse_residual_indices = quant_obj.get("sparse_residual_indices", {})
    sparse_residual_values = quant_obj.get("sparse_residual_values", {})
    lowrank_residual_left = quant_obj.get("lowrank_residual_left", {})
    lowrank_residual_right = quant_obj.get("lowrank_residual_right", {})
    passthrough_orig_dtypes = quant_obj.get("passthrough_orig_dtypes", {})
    for name, q in quant_obj["quantized"].items():
        q_np = np.asarray(q, dtype=np.int8)
        dtype_name = quant_obj["dtypes"][name]
        scale = np.asarray(quant_obj["scales"][name], dtype=np.float32)
        column_scale = (
            np.asarray(awq_scales[name], dtype=np.float32)
            if name in awq_scales
            else None
        )
        out_arr = dequantize_quantized_array(
            q_np,
            scale,
            qmeta.get(name, {}),
            column_scale=column_scale,
        )
        sign = (
            np.asarray(rotation_signs[name], dtype=np.float32)
            if name in rotation_signs
            else None
        )
        out_arr = undo_input_rotation(out_arr, sign)
        out_arr = add_sparse_residual(
            out_arr,
            None if name not in sparse_residual_indices else np.asarray(sparse_residual_indices[name], dtype=np.uint32),
            None if name not in sparse_residual_values else np.asarray(sparse_residual_values[name], dtype=np.float32),
        )
        out_arr = add_lowrank_residual(
            out_arr,
            None if name not in lowrank_residual_left else np.asarray(lowrank_residual_left[name], dtype=np.float32),
            None if name not in lowrank_residual_right else np.asarray(lowrank_residual_right[name], dtype=np.float32),
        )
        out[name] = mx.array(out_arr, dtype=MX_DTYPE_FROM_NAME[dtype_name])
    for name, arr in quant_obj["passthrough"].items():
        # Restore small tensors, undoing the temporary fp16 storage cast if needed.
        out_arr = np.array(arr, copy=True)
        orig_dtype = passthrough_orig_dtypes.get(name)
        if isinstance(orig_dtype, str):
            out[name] = mx.array(out_arr, dtype=MX_DTYPE_FROM_NAME[orig_dtype])
        else:
            out[name] = mx.array(out_arr)
    return out


def build_sentencepiece_luts(
    sp: spm.SentencePieceProcessor, vocab_size: int
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    sp_vocab_size = int(sp.vocab_size())
    table_size = max(sp_vocab_size, vocab_size)
    base_bytes_lut = np.zeros((table_size,), dtype=np.int16)
    has_leading_space_lut = np.zeros((table_size,), dtype=np.bool_)
    is_boundary_token_lut = np.ones((table_size,), dtype=np.bool_)
    for token_id in range(sp_vocab_size):
        if sp.is_control(token_id) or sp.is_unknown(token_id) or sp.is_unused(token_id):
            continue
        is_boundary_token_lut[token_id] = False
        if sp.is_byte(token_id):
            base_bytes_lut[token_id] = 1
            continue
        piece = sp.id_to_piece(token_id)
        if piece.startswith("▁"):
            has_leading_space_lut[token_id] = True
            piece = piece[1:]
        base_bytes_lut[token_id] = len(piece.encode("utf-8"))
    return base_bytes_lut, has_leading_space_lut, is_boundary_token_lut


def validate_dataset_tokenizer_pair(data_path: str, tokenizer_path: str) -> tuple[str, int, int | None]:
    # The shard directory and tokenizer are coupled: val_bpb is only meaningful if we
    # decode bytes with the exact tokenizer that produced the shards. The manifest
    # lets the training script fail fast on accidental dataset/tokenizer mismatches.
    dataset_dir = Path(data_path).resolve()
    actual_train_files = len(list(dataset_dir.glob("fineweb_train_*.bin")))
    if len(dataset_dir.parents) < 2:
        return dataset_dir.name, actual_train_files, None
    manifest_path = dataset_dir.parents[1] / "manifest.json"
    if not manifest_path.is_file():
        return dataset_dir.name, actual_train_files, None

    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    dataset_entry = next((x for x in manifest.get("datasets", []) if x.get("name") == dataset_dir.name), None)
    if dataset_entry is None:
        return dataset_dir.name, actual_train_files, None

    tokenizer_name = dataset_entry.get("tokenizer_name")
    tokenizer_entry = (
        next((x for x in manifest.get("tokenizers", []) if x.get("name") == tokenizer_name), None)
        if tokenizer_name
        else None
    )
    expected_name = Path((tokenizer_entry or {}).get("model_path") or (tokenizer_entry or {}).get("path") or "").name
    if expected_name and Path(tokenizer_path).name != expected_name:
        raise ValueError(f"{dataset_dir.name} expects tokenizer {expected_name}, got {Path(tokenizer_path).name}")
    expected_train_files = (dataset_entry.get("stats") or {}).get("files_train")
    if expected_train_files is not None:
        expected_train_files = int(expected_train_files)
        if actual_train_files > expected_train_files:
            raise ValueError(
                f"{dataset_dir.name} has more train shards than expected: found {actual_train_files}, "
                f"manifest says {expected_train_files}"
            )
    return dataset_dir.name, actual_train_files, expected_train_files


def load_validation_tokens(pattern: str, seq_len: int) -> np.ndarray:
    files = [Path(p) for p in sorted(glob.glob(pattern))]
    if not files:
        raise FileNotFoundError(f"No files found for pattern: {pattern}")
    # The export pipeline writes the fixed first-50k-doc validation set to fineweb_val_*.
    tokens = np.ascontiguousarray(np.concatenate([load_data_shard(file) for file in files], axis=0))
    usable = ((tokens.size - 1) // seq_len) * seq_len
    if usable <= 0:
        raise ValueError(f"Validation split is too short for TRAIN_SEQ_LEN={seq_len}")
    return tokens[: usable + 1]


def select_validation_tokens(tokens: np.ndarray, seq_len: int, subset_seqs: int) -> tuple[np.ndarray, int, int]:
    total_seqs = (tokens.size - 1) // seq_len
    if subset_seqs <= 0:
        return tokens, total_seqs, total_seqs
    selected_seqs = min(subset_seqs, total_seqs)
    if selected_seqs <= 0:
        raise ValueError(f"VAL_SUBSET_SEQS must be positive when set, got {subset_seqs}")
    selected_tokens = np.ascontiguousarray(tokens[: selected_seqs * seq_len + 1])
    return selected_tokens, selected_seqs, total_seqs


def collect_awq_activation_scales(
    args: Hyperparameters,
    model: GPT,
    train_pattern: str,
    dataset_name: str,
    log_fn: Callable[[str], None] | None = None,
) -> dict[str, np.ndarray]:
    awq_requested = AWQ_ENABLED and AWQ_CALIBRATION_SEQS > 0
    lowrank_requested = lowrank_needs_activation_scales() and LOWRANK_CALIBRATION_SEQS > 0
    if not (awq_requested or lowrank_requested):
        return {}
    calibration_seqs = max(
        AWQ_CALIBRATION_SEQS if awq_requested else 0,
        LOWRANK_CALIBRATION_SEQS if lowrank_requested else 0,
    )
    calibration_batch_tokens = max(
        AWQ_CALIBRATION_BATCH_TOKENS if awq_requested else 0,
        LOWRANK_CALIBRATION_BATCH_TOKENS if lowrank_requested else 0,
    )
    batch_tokens = max(
        args.train_seq_len,
        (calibration_batch_tokens // args.train_seq_len) * args.train_seq_len,
    )
    batch_seqs = max(batch_tokens // args.train_seq_len, 1)
    loader = TokenLoader(train_pattern, log_fn=None, dataset_name=dataset_name)
    collector = ActivationAbsMeanCollector()
    global _ACTIVE_ACTIVATION_COLLECTOR
    prev_collector = _ACTIVE_ACTIVATION_COLLECTOR
    _ACTIVE_ACTIVATION_COLLECTOR = collector
    processed_seqs = 0
    try:
        while processed_seqs < calibration_seqs:
            current_seqs = min(batch_seqs, calibration_seqs - processed_seqs)
            x, _ = loader.next_batch(current_seqs * args.train_seq_len, args.train_seq_len)
            hidden = model(x)
            mx.eval(hidden)
            mx.synchronize()
            processed_seqs += current_seqs
            if log_fn is not None and (
                processed_seqs == current_seqs
                or processed_seqs == calibration_seqs
                or processed_seqs % (8 * batch_seqs) == 0
            ):
                log_fn(f"activation_calibration_progress:seqs {processed_seqs}/{calibration_seqs}")
    finally:
        _ACTIVE_ACTIVATION_COLLECTOR = prev_collector
    return collector.finalize()

def loss_and_grad_chunked(
    args: Hyperparameters,
    train_loader: TokenLoader,
    compiled_loss_and_grad,
) -> tuple[mx.array, dict]:
    chunk_sizes = token_chunks(args.microbatch_tokens, args.train_seq_len, args.mlx_max_microbatch_tokens)
    total_tokens = float(sum(chunk_sizes))
    loss_value = mx.array(0.0, dtype=mx.float32)
    grad_accum: dict[str, mx.array] | None = None
    for chunk_tokens in chunk_sizes:
        x, y = train_loader.next_batch(chunk_tokens, args.train_seq_len)
        loss, grads = compiled_loss_and_grad(x, y)
        scale = float(y.size) / total_tokens
        loss_value = loss_value + loss.astype(mx.float32) * scale
        grad_accum = accumulate_flat_grads(grad_accum, grads, scale)
    return loss_value, tree_unflatten(list(grad_accum.items()))


def eval_val(
    args: Hyperparameters,
    compiled_loss,
    val_tokens: np.ndarray,
    base_bytes_lut: np.ndarray,
    has_leading_space_lut: np.ndarray,
    is_boundary_token_lut: np.ndarray,
    log_fn: Callable[[str], None] | None = None,
) -> tuple[float, float]:
    # Validation computes two metrics:
    # - val_loss: token cross-entropy (natural log)
    # - val_bpb: tokenizer-agnostic compression metric used by the challenge
    val_batch_tokens = args.val_batch_size // args.grad_accum_steps
    if val_batch_tokens < args.train_seq_len:
        raise ValueError(
            "VAL_BATCH_SIZE must provide at least one sequence; "
            f"got VAL_BATCH_SIZE={args.val_batch_size}, GRAD_ACCUM_STEPS={args.grad_accum_steps}, "
            f"TRAIN_SEQ_LEN={args.train_seq_len}"
        )
    val_batch_seqs = val_batch_tokens // args.train_seq_len
    total_seqs = (val_tokens.size - 1) // args.train_seq_len
    total_batches = max((total_seqs + val_batch_seqs - 1) // val_batch_seqs, 1)
    total_loss_sum = 0.0
    total_tokens = 0.0
    total_bytes = 0.0
    for batch_idx, batch_seq_start in enumerate(range(0, total_seqs, val_batch_seqs), start=1):
        batch_seq_end = min(batch_seq_start + val_batch_seqs, total_seqs)
        raw_start = batch_seq_start * args.train_seq_len
        raw_end = batch_seq_end * args.train_seq_len + 1
        chunk = val_tokens[raw_start:raw_end]
        x_np = chunk[:-1].reshape(-1, args.train_seq_len)
        y_np = chunk[1:].reshape(-1, args.train_seq_len)
        x = mx.array(x_np, dtype=mx.int32)
        y = mx.array(y_np, dtype=mx.int32)
        chunk_token_count = float(y.size)
        batch_loss = compiled_loss(x, y).astype(mx.float32)
        mx.eval(batch_loss)
        total_loss_sum += float(batch_loss.item()) * chunk_token_count
        prev_ids = x_np.reshape(-1)
        tgt_ids = y_np.reshape(-1)
        bytes_np = base_bytes_lut[tgt_ids].astype(np.int16, copy=True)
        bytes_np += (
            has_leading_space_lut[tgt_ids] & ~is_boundary_token_lut[prev_ids]
        ).astype(np.int16, copy=False)
        total_tokens += chunk_token_count
        total_bytes += float(bytes_np.astype(np.float64).sum())
        if log_fn is not None and total_batches > 1 and (
            batch_idx == 1 or batch_idx == total_batches or batch_idx % 25 == 0
        ):
            log_fn(f"val_progress:{batch_idx}/{total_batches}")
    val_loss = total_loss_sum / total_tokens
    bits_per_token = val_loss / math.log(2.0)
    val_bpb = bits_per_token * (total_tokens / total_bytes)
    return val_loss, val_bpb

def clip_grad_tree(grads_tree: dict, max_norm: float) -> dict:
    if max_norm <= 0:
        return grads_tree
    flat = dict(tree_flatten(grads_tree))
    total_sq = 0.0
    for grad in flat.values():
        total_sq += float(np.sum(np.square(_np_float32(grad)), dtype=np.float64))
    if total_sq <= 0.0:
        return grads_tree
    total_norm = math.sqrt(total_sq)
    if total_norm <= max_norm:
        return grads_tree
    scale = max_norm / (total_norm + 1e-12)
    return tree_unflatten([(k, g * scale) for k, g in flat.items()])

        raise NotImplementedError("train_gpt_mlx.py only supports tied embeddings")
    if not args.tokenizer_path.endswith(".model"):
        raise ValueError(f"TOKENIZER_PATH must point to a SentencePiece .model file: {args.tokenizer_path}")
    sp = spm.SentencePieceProcessor(model_file=args.tokenizer_path)
    if int(sp.vocab_size()) != args.vocab_size:
        raise ValueError(
            f"VOCAB_SIZE={args.vocab_size} does not match tokenizer vocab_size={int(sp.vocab_size())}"
        )
    dataset_name, actual_train_files, expected_train_files = validate_dataset_tokenizer_pair(
        args.data_path,
        args.tokenizer_path,
    )
    val_tokens, selected_val_seqs, total_val_seqs = select_validation_tokens(
        load_validation_tokens(args.val_files, args.train_seq_len),
        args.train_seq_len,
        args.val_subset_seqs,
    )

    base_bytes_lut, has_leading_space_lut, is_boundary_token_lut = build_sentencepiece_luts(
        sp, args.vocab_size
    )

    # ==============================================================================
    # TRAINING SETUP
    # ==============================================================================
    mx.random.seed(args.seed)

    train_loader = TokenLoader(args.train_files, log_fn=log, dataset_name=dataset_name)

    # ==============================================================================
    # MODEL + OPTIMIZER SETUP
    # ==============================================================================
    model = GPT(
        vocab_size=args.vocab_size,
        num_layers=args.num_layers,
        dim=args.model_dim,
        num_heads=args.num_heads,
        num_kv_heads=args.num_kv_heads,
        mlp_mult=args.mlp_mult,
        logit_chunk_tokens=args.logit_chunk_tokens,
        logit_softcap=args.logit_softcap,
        rope_base=args.rope_base,
        tied_embed_init_std=args.tied_embed_init_std,
        qk_gain_init=args.qk_gain_init,
    )
    opt = SplitOptimizers(model, args)

    # ==============================================================================
    # COMPILED TRAIN / EVAL FUNCTIONS (MLX)
    # ==============================================================================
    # The crucial MLX detail is capture scope: this model contains non-trainable arrays too (for example
    # inside RoPE modules), so compiling only against trainable parameters throws "uncaptured inputs".
    # Compiling the model-bound functions and capturing the full model state fixes that while still
    # returning gradients only for trainable parameters via nn.value_and_grad(...).
    compiled_loss = mx.compile(lambda x, y: model.loss(x, y), inputs=model.state, outputs=model.state)
    compiled_loss_and_grad = mx.compile(
        nn.value_and_grad(model, lambda x, y: model.loss(x, y)),
        inputs=model.state,
        outputs=model.state,
    )

    # Print config once so logs are self-describing.
    n_params = sum(int(np.prod(p.shape)) for _, p in tree_flatten(model.parameters()))
    log(f"run_id:{args.run_id}")
    log(f"mlx_version:{mx.__version__}")
    log(f"train_loader:shards pattern={args.train_files}")
    log(f"val_loader:shards pattern={args.val_files} tokens:{val_tokens.size - 1}")
    if selected_val_seqs < total_val_seqs:
        log(
            f"val_loader:subset mode=prefix seqs:{selected_val_seqs}/{total_val_seqs} "
            f"tokens:{val_tokens.size - 1}"
        )
    else:
        log(f"val_loader:subset mode=full seqs:{selected_val_seqs}/{total_val_seqs}")
    if expected_train_files is None:
        log(f"train_loader:dataset:{dataset_name} train_shards:{actual_train_files}")
    elif actual_train_files < expected_train_files:
        log(
            f"WARNING: train_loader:subset dataset:{dataset_name} "
            f"train_shards:{actual_train_files}/{expected_train_files} "
            f"new epochs will arrive sooner than the full dataset"
        )
    else:
        log(f"train_loader:dataset:{dataset_name} train_shards:{actual_train_files}/{expected_train_files}")
    log(f"tokenizer_path:{args.tokenizer_path}")
    log(
        f"model_params:{n_params} vocab_size:{args.vocab_size} layers:{args.num_layers} "
        f"dim:{args.model_dim} heads:{args.num_heads} kv_heads:{args.num_kv_heads} "
        f"seq_len:{args.train_seq_len} tie_embeddings:{args.tie_embeddings}"
    )
    log(
        f"iterations:{args.iterations} train_batch_tokens:{args.train_batch_tokens} grad_accum_steps:{args.grad_accum_steps} "
        f"microbatch_tokens:{args.microbatch_tokens} microbatch_batch_size:{args.microbatch_tokens // args.train_seq_len} "
        f"val_batch_size:{args.val_batch_size} "
        f"warmup_steps:{args.warmup_steps} max_wallclock_seconds:{args.max_wallclock_seconds:.3f}"
    )
    log(f"mlx_max_microbatch_tokens:{args.mlx_max_microbatch_tokens}")
    log(
        f"optimizer:muon+adam muon_matrix_params:{len(opt.matrix_keys)} scalar_params:{len(opt.scalar_keys)} "
        f"embed_lr:{args.tied_embed_lr} "
        f"matrix_lr:{args.matrix_lr} scalar_lr:{args.scalar_lr} "
        f"muon_momentum:{args.muon_momentum} muon_steps:{args.muon_backend_steps}"
    )
    log(f"val_bpb:enabled tokenizer_kind=sentencepiece tokenizer_path={args.tokenizer_path}")
    log(f"compute_dtype:{COMPUTE_DTYPE} compile:True")
    log(
        f"dtypes tok_emb:{model.tok_emb.weight.dtype} "
        f"linear_weight:{model.blocks[0].attn.c_q.weight.dtype} "
        f"skip_weights:{model.skip_weights.dtype}"
    )
    tied_head_mode = (
        f"int{signed_qmax_to_bits(INT8_HEAD_QMAX)}"
        if QUANTIZE_TIED_HEAD
        else "fp16"
    )
    log(
        f"quant_config body_qmax:{INT8_RTN_QMAX} body_bits:{signed_qmax_to_bits(INT8_RTN_QMAX)} "
        f"group_size:{INT8_PER_ROW_GROUP_SIZE} tied_head_mode:{tied_head_mode} codec:lzma "
        f"quant_method:{QUANT_METHOD} input_rotation:{INPUT_ROTATION} scalar_quantizer:{SCALAR_QUANTIZER} "
        f"sparse_residual_frac:{SPARSE_RESIDUAL_FRAC:.6f} "
        f"lowrank_residual_rank:{LOWRANK_RESIDUAL_RANK} lowrank_residual_mode:{LOWRANK_RESIDUAL_MODE}"
    )
    log(
        f"awq_config enabled:{int(AWQ_ENABLED)} calib_seqs:{AWQ_CALIBRATION_SEQS} "
        f"calib_batch_tokens:{AWQ_CALIBRATION_BATCH_TOKENS} scale_tied_head:{int(AWQ_SCALE_TIED_HEAD)} "
        f"alpha_grid:{','.join(f'{alpha:.2f}' for alpha in AWQ_ALPHA_GRID)} "
        f"max_scale:{AWQ_MAX_SCALE:.2f} "
        f"include:{','.join(AWQ_INCLUDE_NAME_PATTERNS) if AWQ_INCLUDE_NAME_PATTERNS else '<all>'} "
        f"exclude:{','.join(AWQ_EXCLUDE_NAME_PATTERNS) if AWQ_EXCLUDE_NAME_PATTERNS else '<none>'}"
    )
    log(
        f"gptq_config enabled:{int(GPTQ_ENABLED)} calib_seqs:{GPTQ_CALIBRATION_SEQS} "
        f"calib_batch_tokens:{GPTQ_CALIBRATION_BATCH_TOKENS} hessian_group_size:{GPTQ_HESSIAN_GROUP_SIZE} "
        f"max_rows_per_observe:{GPTQ_MAX_ROWS_PER_OBSERVE} damp_percent:{GPTQ_DAMP_PERCENT:.4f} "
        f"act_order:{int(GPTQ_ACT_ORDER)}"
    )
    log(
        f"roundopt_config enabled:{int(round_opt_enabled())} method:{ROUND_OPT_METHOD} "
        f"calib_seqs:{ROUND_OPT_CALIBRATION_SEQS} calib_batch_tokens:{ROUND_OPT_CALIBRATION_BATCH_TOKENS} "
        f"max_rows_per_observe:{ROUND_OPT_MAX_ROWS_PER_OBSERVE} "
        f"scale_grid:{','.join(f'{value:.3f}' for value in ROUND_OPT_SCALE_GRID)} "
        f"bias_grid:{','.join(f'{value:.4f}' for value in ROUND_OPT_BIAS_GRID)} "
        f"use_full_hessian:{int(ROUND_OPT_USE_FULL_HESSIAN)} rowwise_selection:{int(ROUND_OPT_ROWWISE_SELECTION)}"
    )
    log(
        f"lowrank_config enabled:{int(lowrank_residual_enabled())} rank:{LOWRANK_RESIDUAL_RANK} "
        f"mode:{LOWRANK_RESIDUAL_MODE} calib_seqs:{LOWRANK_CALIBRATION_SEQS} "
        f"calib_batch_tokens:{LOWRANK_CALIBRATION_BATCH_TOKENS}"
    )

    # ==============================================================================
    # TRAINING LOOP
    # ==============================================================================
    if args.warmup_steps > 0:
        # Warmup should only prime MLX compile/allocation paths. Updating parameters here forces us
        # to snapshot and restore model/optimizer state, which is expensive on unified-memory Macs.
        # Instead we run the real train shapes, force the loss/grads to materialize, and then reset
        # the loader so measured training still starts from the true init and token window.
        for warmup_step in range(args.warmup_steps):
            accum: dict[str, mx.array] | None = None
            warmup_loss = mx.array(0.0, dtype=mx.float32)
            grad_scale = 1.0 / args.grad_accum_steps
            for _ in range(args.grad_accum_steps):
                warmup_loss, grads = loss_and_grad_chunked(args, train_loader, compiled_loss_and_grad)
                accum = accumulate_flat_grads(accum, grads, grad_scale)
            mx.eval(warmup_loss, accum)
            mx.synchronize()
            if args.warmup_steps <= 20 or (warmup_step + 1) % 10 == 0 or warmup_step + 1 == args.warmup_steps:
                log(f"warmup_step:{warmup_step + 1}/{args.warmup_steps}")

        # Prime the standalone eval graph once too. It is compiled separately from value_and_grad.
        val_batch_tokens = args.val_batch_size // args.grad_accum_steps
        if val_batch_tokens < args.train_seq_len:
            raise ValueError(
                "VAL_BATCH_SIZE must provide at least one sequence; "
                f"got VAL_BATCH_SIZE={args.val_batch_size}, GRAD_ACCUM_STEPS={args.grad_accum_steps}, "
                f"TRAIN_SEQ_LEN={args.train_seq_len}"
            )
        warm_val_seqs = min(val_batch_tokens // args.train_seq_len, (val_tokens.size - 1) // args.train_seq_len)
        warm_chunk = val_tokens[: warm_val_seqs * args.train_seq_len + 1]
        x_val = mx.array(warm_chunk[:-1].reshape(-1, args.train_seq_len), dtype=mx.int32)
        y_val = mx.array(warm_chunk[1:].reshape(-1, args.train_seq_len), dtype=mx.int32)
        warm_val_loss = compiled_loss(x_val, y_val)
        mx.eval(warm_val_loss)
        mx.synchronize()

        train_loader = TokenLoader(args.train_files, log_fn=log, dataset_name=dataset_name)

    train_time_ms = 0.0
    max_wallclock_ms = 1000.0 * args.max_wallclock_seconds if args.max_wallclock_seconds > 0 else None
    stop_after_step: int | None = None
    t0 = time.perf_counter()
    step = 0
    while True:
        last_step = step == args.iterations or (stop_after_step is not None and step >= stop_after_step)
        should_run_periodic_val = args.val_loss_every > 0 and step % args.val_loss_every == 0 and not last_step
        should_run_final_float_val = last_step and not args.skip_final_float_val
        if last_step or should_run_periodic_val:
            train_time_ms += 1000.0 * (time.perf_counter() - t0)
            if should_run_periodic_val or should_run_final_float_val:
                # Validation always scans the same fixed token prefix selected at startup.
                val_loss, val_bpb = eval_val(
                    args,
                    compiled_loss,
                    val_tokens,
                    base_bytes_lut,
                    has_leading_space_lut,
                    is_boundary_token_lut,
                    log_fn=log,
                )
                if step % 25 == 0 or last_step:
                    log(
                        f"step:{step}/{args.iterations} val_loss:{val_loss:.4f} val_bpb:{val_bpb:.4f} "
                        f"train_time:{train_time_ms:.0f}ms step_avg:{train_time_ms / max(step, 1):.2f}ms"
                    )
            elif last_step:
                log(f"skipping_final_float_val:1 train_time:{train_time_ms:.0f}ms step:{step}/{args.iterations}")
            t0 = time.perf_counter()
        if last_step:
            if stop_after_step is not None and step < args.iterations:
                log(f"stopping_early: wallclock_cap train_time:{train_time_ms:.0f}ms step:{step}/{args.iterations}")
            break

        lr_mul = args.lr_mul(step, train_time_ms + 1000.0 * (time.perf_counter() - t0))
        step_t0 = time.perf_counter()

        accum: dict[str, mx.array] | None = None
        train_loss = mx.array(0.0, dtype=mx.float32)
        grad_scale = 1.0 / args.grad_accum_steps
        for _ in range(args.grad_accum_steps):
            loss, grads = loss_and_grad_chunked(args, train_loader, compiled_loss_and_grad)
            accum = accumulate_flat_grads(accum, grads, grad_scale)
            train_loss = train_loss + loss.astype(mx.float32) * grad_scale

        grads = tree_unflatten(list(accum.items()))
        grads = clip_grad_tree(grads, args.grad_clip_norm)
        train_loss_value = float(train_loss.item())
        opt.step(model, grads, step=step, lr_mul=lr_mul)
        mx.synchronize()

        step_ms = 1000.0 * (time.perf_counter() - step_t0)
        approx_train_time_ms = train_time_ms + 1000.0 * (time.perf_counter() - t0)
        tok_s = args.train_batch_tokens / (step_ms / 1000.0)
        step += 1
        if args.train_log_every > 0 and (step <= 10 or step % args.train_log_every == 0 or stop_after_step is not None):
            log(
                f"step:{step}/{args.iterations} train_loss:{train_loss_value:.4f} "
                f"train_time:{approx_train_time_ms:.0f}ms step_avg:{approx_train_time_ms / step:.2f}ms tok_s:{tok_s:.0f}"
            )
        if max_wallclock_ms is not None and stop_after_step is None and approx_train_time_ms >= max_wallclock_ms:
            stop_after_step = step

    train_peak_rss = peak_rss_bytes()
    log(f"train_peak_rss_bytes:{train_peak_rss} train_peak_rss:{format_bytes(train_peak_rss)}")

    # ==============================================================================
    # FINAL SERIALIZATION + QUANTIZED ROUNDTRIP EVAL
    # ==============================================================================
    # We always write a raw artifact and a quantized artifact, then validate the
    # quantized roundtrip directly by loading the dequantized tensors back into the
    # model and running one final validation pass.
    out_path = out_dir / f"{args.run_id}_mlx_model.npz"
    flat_state = {k: v for k, v in tree_flatten(model.state)}
    mx.savez(str(out_path), **flat_state)
    log(f"saved_model:{out_path} bytes:{out_path.stat().st_size}")

    awq_activation_scales = collect_awq_activation_scales(
        args,
        model,
        args.train_files,
        dataset_name,
        log_fn=log,
    )
    gptq_hessian_groups = collect_gptq_hessian_groups(
        args,
        model,
        args.train_files,
        dataset_name,
        log_fn=log,
    )
    roundopt_hessian_groups = collect_roundopt_hessian_groups(
        args,
        model,
        args.train_files,
        dataset_name,
        log_fn=log,
    )
    if AWQ_ENABLED:
        log(f"awq_calibration_tensors:{len(awq_activation_scales)}")
    if lowrank_needs_activation_scales():
        log(f"lowrank_activation_tensors:{len(awq_activation_scales)}")
    if GPTQ_ENABLED:
        log(f"gptq_calibration_tensors:{len(gptq_hessian_groups)}")
    if round_opt_enabled():
        log(f"roundopt_calibration_tensors:{len(roundopt_hessian_groups)}")

    quant_obj, quant_stats = quantize_state_dict_int8(
        flat_state,
        awq_activation_scales=awq_activation_scales,
        gptq_hessian_groups=gptq_hessian_groups,
        roundopt_hessian_groups=roundopt_hessian_groups,
        log_fn=log,
    )
    quant_raw = pickle.dumps(quant_obj, protocol=pickle.HIGHEST_PROTOCOL)
    quant_blob = lzma.compress(quant_raw, preset=6)
    quant_serialized_bytes = len(quant_raw)
    quant_path = out_dir / f"{args.run_id}_mlx_model.int8.xz"
    with quant_path.open("wb") as f:
        f.write(quant_blob)
    quant_file_bytes = quant_path.stat().st_size
    ratio = quant_stats["baseline_tensor_bytes"] / max(quant_stats["int8_payload_bytes"], 1)
    log(
        f"serialized_model_int8_zlib:{quant_file_bytes} bytes "
        f"(codec:lzma payload:{quant_stats['int8_payload_bytes']} raw_pickle:{quant_serialized_bytes} "
        f"payload_ratio:{ratio:.2f}x awq_tensors:{quant_stats['num_awq_tensors']} "
        f"awq_scale_bytes:{quant_stats['awq_scale_bytes']} gptq_tensors:{quant_stats['num_gptq_tensors']} "
        f"roundopt_tensors:{quant_stats['num_roundopt_tensors']} "
        f"rotated_tensors:{quant_stats['num_rotated_tensors']} rotation_bytes:{quant_stats['rotation_bytes']} "
        f"sparse_tensors:{quant_stats['num_sparse_residual_tensors']} "
        f"sparse_values:{quant_stats['sparse_residual_values']} "
        f"sparse_bytes:{quant_stats['sparse_residual_bytes']} "
        f"lowrank_tensors:{quant_stats['num_lowrank_residual_tensors']} "
        f"lowrank_rank_sum:{quant_stats['lowrank_residual_rank_sum']} "
        f"lowrank_bytes:{quant_stats['lowrank_residual_bytes']})"
    )
    log(f"artifact_bytes_int8_zlib:{quant_file_bytes + code_bytes}")

    with quant_path.open("rb") as f:
        quant_blob_disk = f.read()
    quant_flat = dequantize_state_dict_int8(pickle.loads(lzma.decompress(quant_blob_disk)))
    model.update(tree_unflatten(list(quant_flat.items())))
    q_t0 = time.perf_counter()
    q_val_loss, q_val_bpb = eval_val(
        args,
        compiled_loss,
        val_tokens,
        base_bytes_lut,
        has_leading_space_lut,
        is_boundary_token_lut,
        log_fn=log,
    )
    q_eval_ms = 1000.0 * (time.perf_counter() - q_t0)
    log(f"final_int8_zlib_roundtrip val_loss:{q_val_loss:.4f} val_bpb:{q_val_bpb:.4f} eval_time:{q_eval_ms:.0f}ms")
    log(f"final_int8_zlib_roundtrip_exact val_loss:{q_val_loss:.8f} val_bpb:{q_val_bpb:.8f}")


if __name__ == "__main__":
    main()
